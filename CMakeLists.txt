cmake_minimum_required(VERSION 3.20)
project(atlas VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Boost 1.74 REQUIRED)
find_package(Threads REQUIRED)
set(ATLAS_SANITIZER "" CACHE STRING "address, undefined, thread, or empty")
if(ATLAS_SANITIZER)
  add_compile_options(-fsanitize=${ATLAS_SANITIZER} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${ATLAS_SANITIZER})
endif()
add_library(atlas_core src/core.cpp src/service.cpp src/protocol.cpp)
target_include_directories(atlas_core PUBLIC include ${Boost_INCLUDE_DIRS})
target_compile_definitions(atlas_core PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
target_link_libraries(atlas_core PUBLIC Threads::Threads)
target_compile_options(atlas_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(atlas_server src/main.cpp src/server.cpp src/session.cpp)
target_link_libraries(atlas_server PRIVATE atlas_core)
target_compile_options(atlas_server PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
include(CTest)
if(BUILD_TESTING)
  add_executable(atlas_unit tests/unit.cpp)
  target_link_libraries(atlas_unit PRIVATE atlas_core)
  add_test(NAME unit COMMAND atlas_unit)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py --server $<TARGET_FILE:atlas_server>)
  set_tests_properties(integration PROPERTIES TIMEOUT 180)
endif()
install(TARGETS atlas_server RUNTIME DESTINATION bin)
install(FILES config/server.conf DESTINATION etc/atlas)
